"""System prompt builder for AI admin."""
import json
from database.models import Business


def build_system_prompt(business: Business) -> str:
    """Build dynamic system prompt based on current business state."""
    try:
        photos_count = len(json.loads(business.photos)) if business.photos else 0
    except Exception:
        photos_count = 0

    has_video = bool(business.video)
    has_directions = bool(business.directions)

    break_info = "нет"
    if business.break_start and business.break_end:
        break_info = f"{business.break_start.strftime('%H:%M')}–{business.break_end.strftime('%H:%M')}"

    prepayment_info = "выключена"
    if business.prepayment_enabled:
        prepayment_info = f"включена {business.prepayment_percent}%"

    wh_start = business.working_hours_start.strftime('%H:%M') if business.working_hours_start else "?"
    wh_end = business.working_hours_end.strftime('%H:%M') if business.working_hours_end else "?"

    settings_block = (
        f"━━━ ТЕКУЩИЕ НАСТРОЙКИ БИЗНЕСА ━━━\n"
        f"ID: {business.id}\n"
        f"Название: {business.name}\n"
        f"Описание: {business.description or 'не задано'}\n"
        f"Адрес: {business.address or 'не задан'}\n"
        f"Телефон: {business.phone or 'не задан'}\n"
        f"Часы работы: {wh_start}–{wh_end}\n"
        f"Перерыв: {break_info}\n"
        f"Фото в галерее: {photos_count}\n"
        f"Видео: {'есть' if has_video else 'нет'}\n"
        f"Схема проезда: {'есть' if has_directions else 'нет'}\n"
        f"Предоплата: {prepayment_info}\n"
        f"Реквизиты: {business.payment_details or 'не заданы'}"
    )

    instruction_block = ""
    if business.ai_instruction:
        instruction_block = (
            "\n\n━━━ ИНСТРУКЦИЯ ПО ЭТОМУ БИЗНЕСУ ━━━\n"
            + business.ai_instruction
        )

    return (
        f"Ты — AI-администратор бизнеса «{business.name}» в Telegram-боте для онлайн-записи.\n"
        "Помогаешь владельцу управлять настройками, записями и коммуникацией через естественный диалог.\n\n"
        f"{settings_block}"
        f"{instruction_block}\n\n"
        "━━━ ПРАВИЛА ━━━\n"
        "- Хочет изменить настройку — сразу вызывай функцию, НЕ проси словесного подтверждения (придут кнопки).\n"
        "- Хочет записи — вызывай get_bookings() с нужными фильтрами.\n"
        "- Прислал фото с подписью — определи по подписи что делать и вызови нужную функцию.\n"
        "- Непонятный запрос — задай ОДИН короткий уточняющий вопрос.\n"
        "- Перед рассылкой — сообщи сколько получателей.\n"
        "- Отвечай кратко, без предисловий.\n"
        "- Язык: русский, стиль деловой но дружелюбный.\n"
        "- Работаешь только с этим бизнесом. На вопросы не по теме: «Я работаю только с настройками бизнеса»."

```python
"""AI Admin Service — история диалога + вызов OpenRouter API."""
import asyncio
import json
import logging
from typing import Optional

import aiohttp

from bot.config import OPENROUTER_AI_ADMIN_KEY

logger = logging.getLogger(__name__)

MODEL           = "stepfun/step-3.5-flash:free"
OPENROUTER_URL  = "https://openrouter.ai/api/v1/chat/completions"
MAX_OWNER_MSGS  = 15   # компакция после этого числа сообщений от владельца

_histories:       dict[int, list[dict]] = {}
_lock             = asyncio.Lock()
_pending_actions: dict[str, dict] = {}   # action_id -> {name, args, business_id, file_id}


# ─── ИСТОРИЯ ──────────────────────────────────────────────────────────────────

async def get_history(user_id: int) -> list[dict]:
    async with _lock:
        return list(_histories.get(user_id, []))


async def add_to_history(user_id: int, role: str, content) -> None:
    async with _lock:
        _histories.setdefault(user_id, []).append({"role": role, "content": content})


async def clear_history(user_id: int) -> None:
    async with _lock:
        _histories.pop(user_id, None)


def _count_user_msgs(history: list[dict]) -> int:
    return sum(1 for m in history if m["role"] == "user")


async def _compact(user_id: int, system_prompt: str) -> None:
    """Сжать историю в 3-5 предложений."""
    async with _lock:
        history = list(_histories.get(user_id, []))
    if not history:
        return
    messages = [
        {"role": "system", "content": system_prompt},
        *history,
        {"role": "user", "content": (
            "Сожми эту историю диалога в 3-5 предложений на русском. "
            "Сохрани ключевые факты: что менялось, что решили. "
            "Только сжатый текст, без предисловий."
        )},
    ]
    try:
        summary = await _call_api(messages, tools=None)
        if isinstance(summary, str):
            async with _lock:
                _histories[user_id] = [
                    {"role": "assistant",
                     "content": f"[Сводка предыдущего разговора]\n{summary}"}
                ]
            logger.info("AI admin history compacted for user %s", user_id)
    except Exception as exc:
        logger.error("Compaction failed for user %s: %s", user_id, exc)
        async with _lock:
            _histories.pop(user_id, None)


# ─── OPENROUTER ───────────────────────────────────────────────────────────────

async def _call_api(messages: list[dict], tools: Optional[list[dict]]) -> "str | dict":
    """
    Вызов OpenRouter.
    Возвращает str (текст) или dict (tool call).
    """
    headers = {
        "Authorization": f"Bearer {OPENROUTER_AI_ADMIN_KEY}",
        "Content-Type":  "application/json",
        "HTTP-Referer":  "https://booking-bot.local",
    }
    payload: dict = {"model": MODEL, "messages": messages, "temperature": 0.3}
    if tools:
        payload["tools"]       = tools
        payload["tool_choice"] = "auto"

    async with aiohttp.ClientSession() as session:
        async with session.post(
            OPENROUTER_URL, headers=headers, json=payload,
            timeout=aiohttp.ClientTimeout(total=45),
        ) as resp:
            if resp.status != 200:
                raise RuntimeError(f"OpenRouter {resp.status}: {(await resp.text())[:300]}")
            data = await resp.json()

    msg = data["choices"][0]["message"]
    if msg.get("tool_calls"):
        tc = msg["tool_calls"][0]
        return {
            "name":        tc["function"]["name"],
            "arguments":   json.loads(tc["function"]["arguments"]),
            "tool_call_id": tc["id"],
            "raw_message": msg,
        }
    return msg.get("content") or ""


# ─── ОСНОВНАЯ ТОЧКА ВХОДА ─────────────────────────────────────────────────────

async def process_message(
    user_id:       int,
    system_prompt: str,
    user_content,           # str или list (multimodal)
    tools:         list[dict],
) -> "str | dict":
    await add_to_history(user_id, "user", user_content)
    history  = await get_history(user_id)
    messages = [{"role": "system", "content": system_prompt}] + history

    result = await _call_api(messages, tools)

    if isinstance(result, str):
        await add_to_history(user_id, "assistant", result)
    else:
        raw_content = result["raw_message"].get("content") or ""
        await add_to_history(user_id, "assistant", raw_content)

    updated = await get_history(user_id)
    if _count_user_msgs(updated) >= MAX_OWNER_MSGS:
        asyncio.create_task(_compact(user_id, system_prompt))

    return result


# ─── ОЖИДАЮЩИЕ ДЕЙСТВИЯ ───────────────────────────────────────────────────────

def store_pending_action(
    action_id:   str,
    name:        str,
    args:        dict,
    business_id: int,
    file_id:     Optional[str] = None,
) -> None:
    _pending_actions[action_id] = {
        "name": name, "args": args,
        "business_id": business_id, "file_id": file_id,
    }


def pop_pending_action(action_id: str) -> Optional[dict]:
    return _pending_actions.pop(action_id, None)
```

async def execute_tool(
    tool_name:   str,
    args:        dict,
    business_id: int,
    bot=None,
    file_id:     Optional[str] = None,
) -> str:
    async with async_session_maker() as session:
        try:
            result = await _exec(session, tool_name, args, business_id, bot, file_id)
            await session.commit()
            return result
        except Exception as exc:
            await session.rollback()
            logger.error("Tool %s failed: %s", tool_name, exc, exc_info=True)
            return f"❌ Ошибка: {exc}"


async def _exec(
    s: AsyncSession, name: str, args: dict,
    biz_id: int, bot, file_id: Optional[str],
) -> str:

    if name == "update_name":
        await _upd(s, biz_id, name=args["new_name"])
        return f"✅ Название: «{args['new_name']}»"

    if name == "update_description":
        await _upd(s, biz_id, description=args["new_description"])
        return "✅ Описание обновлено"

    if name == "update_address":
        await _upd(s, biz_id, address=args["new_address"])
        return f"✅ Адрес: {args['new_address']}"

    if name == "update_phone":
        await _upd(s, biz_id, phone=args["new_phone"])
        return f"✅ Телефон: {args['new_phone']}"

    if name == "update_working_hours":
        await _upd(s, biz_id,
                   working_hours_start=_t(args["start"]),
                   working_hours_end=_t(args["end"]))
        return f"✅ Часы работы: {args['start']}–{args['end']}"

    if name == "update_break_time":
        if args.get("start") and args.get("end"):
            await _upd(s, biz_id, break_start=_t(args["start"]), break_end=_t(args["end"]))
            return f"✅ Перерыв: {args['start']}–{args['end']}"
        await _upd(s, biz_id, break_start=None, break_end=None)
        return "✅ Перерыв убран"

    if name == "update_directions":
        await _upd(s, biz_id, directions=args["text"])
        return "✅ Схема проезда обновлена"

    if name == "delete_all_photos":
        await _upd(s, biz_id, photos="[]")
        return "✅ Все фотографии удалены"

    if name == "add_photo":
        if not file_id:
            return "❌ Фото не найдено в сообщении"
        b = await _biz(s, biz_id)
        try:
            photos = json.loads(b.photos) if b.photos else []
        except Exception:
            photos = []
        photos.append(file_id)
        await _upd(s, biz_id, photos=json.dumps(photos))
        return f"✅ Фото добавлено (всего: {len(photos)})"

    if name == "update_video":
        if not file_id:
            return "❌ Видео не найдено в сообщении"
        await _upd(s, biz_id, video=file_id)
        return "✅ Видео обновлено"

    if name == "update_cancellation_photo":
        if not file_id:
            return "❌ Фото не найдено в сообщении"
        await _upd(s, biz_id, cancellation_policy_photo=file_id)
        return "✅ Фото политики отмены обновлено"

    if name == "toggle_prepayment":
        await _upd(s, biz_id, prepayment_enabled=args["enabled"])
        return f"✅ Предоплата {'включена' if args['enabled'] else 'выключена'}"

    if name == "update_prepayment_settings":
        upd: dict = {"prepayment_percent": args["percent"]}
        if args.get("phone"):   upd["payment_phone"]   = args["phone"]
        if args.get("details"): upd["payment_details"] = args["details"]
        await _upd(s, biz_id, **upd)
        return "✅ Настройки предоплаты обновлены"

    if name == "update_ai_instruction":
        await _upd(s, biz_id, ai_instruction=args["instruction"])
        return "✅ Инструкция AI обновлена"

    if name == "get_current_settings":
        return _fmt(await _biz(s, biz_id))

    if name == "get_bookings":
        return await _bookings(s, biz_id, args)

    if name == "cancel_booking":
        bk = await _booking(s, args["booking_id"], biz_id)
        if not bk: return "❌ Запись не найдена"
        bk.status = "cancelled"
        if args.get("reason"):
            bk.notes = (bk.notes or "") + f"\nОтменено: {args['reason']}"
        await _notify(bot, bk.client.telegram_id,
                      f"❌ Ваша запись на {bk.booking_date} в {bk.booking_time.strftime('%H:%M')} отменена."
                      + (f"\nПричина: {args['reason']}" if args.get("reason") else ""))
        return f"✅ Запись #{args['booking_id']} отменена"

    if name == "reschedule_booking":
        bk = await _booking(s, args["booking_id"], biz_id)
        if not bk: return "❌ Запись не найдена"
        bk.booking_date = datetime.date.fromisoformat(args["new_date"])
        bk.booking_time = _t(args["new_time"])
        await _notify(bot, bk.client.telegram_id,
                      f"📅 Ваша запись перенесена на {args['new_date']} в {args['new_time']}")
        return f"✅ Запись #{args['booking_id']} перенесена"

    if name == "add_booking_note":
        bk = await _booking(s, args["booking_id"], biz_id)
        if not bk: return "❌ Запись не найдена"
        bk.notes = args["note"]
        return f"✅ Заметка добавлена к #{args['booking_id']}"

    if name == "confirm_booking":
        bk = await _booking(s, args["booking_id"], biz_id)
        if not bk: return "❌ Запись не найдена"
        bk.status = "confirmed"
        return f"✅ Запись #{args['booking_id']} подтверждена"

    if name == "send_message_to_client":
        bk = await _booking(s, args["booking_id"], biz_id)
        if not bk: return "❌ Запись не найдена"
        if not bot: return "❌ Бот недоступен"
        if bk.client.telegram_id <= 0: return "❌ У клиента нет Telegram"
        await bot.send_message(bk.client.telegram_id, args["text"])
        return f"✅ Сообщение отправлено {bk.client.full_name or 'клиенту'}"

    if name == "send_broadcast":
        return await _broadcast(s, biz_id, args["text"], bot)

    return f"❓ Неизвестная функция: {name}"


# ─── HELPERS ──────────────────────────────────────────────────────────────────

def _t(hm: str):
    return datetime.datetime.strptime(hm, "%H:%M").time()


async def _upd(s: AsyncSession, biz_id: int, **fields) -> None:
    res = await s.execute(select(Business).where(Business.id == biz_id))
    biz = res.scalar_one_or_none()
    if not biz: raise ValueError(f"Бизнес {biz_id} не найден")
    for k, v in fields.items(): setattr(biz, k, v)


async def _biz(s: AsyncSession, biz_id: int) -> Business:
    res = await s.execute(select(Business).where(Business.id == biz_id))
    return res.scalar_one_or_none()


async def _booking(s: AsyncSession, booking_id: int, biz_id: int):
    res = await s.execute(
        select(Booking)
        .options(selectinload(Booking.client), selectinload(Booking.service))
        .where(and_(Booking.id == booking_id, Booking.business_id == biz_id))
    )
    return res.scalar_one_or_none()


async def _notify(bot, telegram_id: int, text: str) -> None:
    if bot and telegram_id and telegram_id > 0:
        try:
            await bot.send_message(telegram_id, text)
        except Exception:
            pass


async def _bookings(s: AsyncSession, biz_id: int, args: dict) -> str:
    q = (
        select(Booking)
        .options(selectinload(Booking.client), selectinload(Booking.service))
        .where(Booking.business_id == biz_id)
    )
    if args.get("date"):
        q = q.where(Booking.booking_date == datetime.date.fromisoformat(args["date"]))
    if args.get("status") and args["status"] != "all":
        q = q.where(Booking.status == args["status"])
    q = q.order_by(Booking.booking_date, Booking.booking_time).limit(20)
    rows = (await s.execute(q)).scalars().all()
    if not rows:
        return "📭 Записей не найдено"
    icons = {"pending": "⏳", "confirmed": "✅", "cancelled": "❌", "completed": "✔️"}
    lines = [f"📋 Записей: {len(rows)}\n"]
    for b in rows:
        cname = b.client.full_name or b.client.username or "Без имени"
        if args.get("client_name") and args["client_name"].lower() not in cname.lower():
            continue
        svc = b.service.name if b.service else "—"
        lines.append(f"{icons.get(b.status,'❓')} #{b.id} | {b.booking_date} {b.booking_time.strftime('%H:%M')} | {cname} | {svc}")
    return "\n".join(lines)


async def _broadcast(s: AsyncSession, biz_id: int, text: str, bot) -> str:
    if not bot: return "❌ Бот недоступен"
    res = await s.execute(
        select(User)
        .join(Booking, Booking.client_id == User.id)
        .where(and_(Booking.business_id == biz_id, User.telegram_id > 0))
        .distinct()
    )
    clients = res.scalars().all()
    if not clients: return "❌ Нет клиентов для рассылки"
    sent = failed = 0
    for c in clients:
        try:
            await bot.send_message(c.telegram_id, text)
            sent += 1
        except Exception:
            failed += 1
    return f"✅ Рассылка: {sent} получателей" + (f", {failed} ошибок" if failed else "")


def _fmt(b: Business) -> str:
    if not b: return "❌ Бизнес не найден"
    try:
        photos = len(json.loads(b.photos)) if b.photos else 0
    except Exception:
        photos = 0
    return (
        f"📋 Настройки «{b.name}»\n\n"
        f"📝 {b.description or '—'}\n"
        f"📍 {b.address or '—'}\n"
        f"📞 {b.phone or '—'}\n"
        f"⏰ {b.working_hours_start}–{b.working_hours_end}\n"
        f"☕ Перерыв: {b.break_start or '—'}\n"
        f"📷 Фото: {photos}\n"
        f"🎬 Видео: {'есть' if b.video else 'нет'}\n"
        f"🗺 Проезд: {'есть' if b.directions else 'нет'}\n"
        f"💳 Предоплата: {'вкл ' + str(b.prepayment_percent) + '%' if b.prepayment_enabled else 'выкл'}"
    )
```

"""
AI Admin catch-all handler.
Перехватывает свободный текст и медиа от владельцев бизнесов
когда нет активного FSM-состояния.
"""
import logging
import uuid
from typing import Optional

from aiogram import Router, F, Bot
from aiogram.filters import StateFilter
from aiogram.types import Message, CallbackQuery
from sqlalchemy import select

from database.database import async_session_maker
from database.models import Business, User
from bot.prompts.business_admin import build_system_prompt
from bot.services.ai_admin import process_message, store_pending_action, pop_pending_action
from bot.services.ai_admin_tools import TOOL_SCHEMAS, READ_ONLY_TOOLS, describe_tool_call
from bot.services.ai_admin_executor import execute_tool
from bot.keyboards.ai_admin_kb import confirm_keyboard

router = Router()
logger = logging.getLogger(__name__)


async def _owner_business(telegram_id: int) -> Optional[Business]:
    """Вернуть первый активный бизнес владельца или None."""
    async with async_session_maker() as session:
        res = await session.execute(select(User).where(User.telegram_id == telegram_id))
        user = res.scalar_one_or_none()
        if not user:
            return None
        res = await session.execute(
            select(Business).where(Business.owner_id == user.id).limit(1)
        )
        return res.scalar_one_or_none()


async def _fresh(business_id: int) -> Optional[Business]:
    async with async_session_maker() as session:
        res = await session.execute(select(Business).where(Business.id == business_id))
        return res.scalar_one_or_none()


@router.message(StateFilter(None), F.content_type.in_({"text", "photo", "document"}))
async def ai_message(message: Message, bot: Bot):
    # Пропустить команды
    if message.text and message.text.startswith("/"):
        return

    biz = await _owner_business(message.from_user.id)
    if not biz:
        return

    # Определить контент и file_id
    file_id: Optional[str] = None
    if message.photo:
        file_id     = message.photo[-1].file_id
        caption     = message.caption or "фото без описания"
        user_content = f"Владелец прислал фото (file_id={file_id}). Подпись: {caption}"
    elif message.document:
        file_id      = message.document.file_id
        caption      = message.caption or "файл без описания"
        user_content = (
            f"Владелец прислал файл (file_id={file_id}, "
            f"name={message.document.file_name}). Подпись: {caption}"
        )
    else:
        user_content = message.text

    fresh_biz     = await _fresh(biz.id)
    system_prompt = build_system_prompt(fresh_biz)

    await bot.send_chat_action(message.chat.id, "typing")

    try:
        result = await process_message(
            user_id=message.from_user.id,
            system_prompt=system_prompt,
            user_content=user_content,
            tools=TOOL_SCHEMAS,
        )
    except Exception as exc:
        logger.error("AI admin error (user=%s): %s", message.from_user.id, exc, exc_info=True)
        await message.answer("⚠️ Ошибка AI. Попробуйте ещё раз.")
        return

    if isinstance(result, str):
        await message.answer(result)
        return

    tool_name = result["name"]
    tool_args = result["arguments"]

    # Read-only — без подтверждения
    if tool_name in READ_ONLY_TOOLS:
        await message.answer(await execute_tool(tool_name, tool_args, biz.id, bot, file_id))
        return

    # Показать карточку подтверждения
    action_id = str(uuid.uuid4())[:8]
    store_pending_action(action_id, tool_name, tool_args, biz.id, file_id)
    await message.answer(
        f"🤖 Предлагаю:\n\n{describe_tool_call(tool_name, tool_args)}",
        reply_markup=confirm_keyboard(action_id),
    )


@router.callback_query(F.data.startswith("ai_confirm:"))
async def ai_confirm(callback: CallbackQuery, bot: Bot):
    action_id = callback.data.split(":", 1)[1]
    action    = pop_pending_action(action_id)
    if not action:
        await callback.answer("Действие устарело", show_alert=True)
        await callback.message.edit_reply_markup(reply_markup=None)
        return
    await callback.message.edit_reply_markup(reply_markup=None)
    result = await execute_tool(
        action["name"], action["args"],
        action["business_id"], bot, action.get("file_id"),
    )
    await callback.message.answer(result)
    await callback.answer()


@router.callback_query(F.data == "ai_cancel")
async def ai_cancel(callback: CallbackQuery):
    await callback.message.edit_reply_markup(reply_markup=None)
    await callback.answer("Отменено")
```
